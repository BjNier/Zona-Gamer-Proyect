    <script>
        document.addEventListener("DOMContentLoaded", function() {
            const sections = document.querySelectorAll('.unique-section');
            let currentSectionIndex = 0;

            function setActiveSection(index) {
                sections.forEach((section, idx) => {
                    if (idx === index) {
                        section.classList.add('active');
                    } else {
                        section.classList.remove('active');
                    }
                });
            }

            // Set initial active section based on scroll position
            function setInitialSection() {
                const scrollPosition = window.scrollY;
                const windowHeight = window.innerHeight;

                sections.forEach((section, index) => {
                    const sectionTop = section.offsetTop;
                    const sectionHeight = section.clientHeight;

                    if (scrollPosition >= sectionTop - windowHeight / 2 && scrollPosition < sectionTop + sectionHeight / 2) {
                        currentSectionIndex = index;
                    }
                });

                setActiveSection(currentSectionIndex);
            }

            setInitialSection();

            window.addEventListener('scroll', () => {
                setInitialSection();
            });

            window.addEventListener('wheel', (event) => {
                if (event.deltaY < 0) {
                    // Scrolling up
                    currentSectionIndex = Math.max(currentSectionIndex - 1, 0);
                } else {
                    // Scrolling down
                    currentSectionIndex = Math.min(currentSectionIndex + 1, sections.length - 1);
                }

                setActiveSection(currentSectionIndex);

                // Prevent default scroll behavior
                event.preventDefault();
            });

            window.addEventListener('resize', () => {
                setInitialSection();
            });

            // Galería de imágenes para la sección 1
            const galleryImages1 = document.querySelectorAll('#unique-section1 .gallery img');
            let currentImageIndex1 = 0;

            function changeImage1() {
                galleryImages1[currentImageIndex1].classList.remove('active');
                currentImageIndex1 = (currentImageIndex1 + 1) % galleryImages1.length;
                galleryImages1[currentImageIndex1].classList.add('active');
            }

            setInterval(changeImage1, 3000); // Cambia de imagen cada 3 segundos

            // Galería de imágenes para la sección 2
            const galleryImages2 = document.querySelectorAll('#unique-section2 .gallery img');
            let currentImageIndex2 = 0;

            function changeImage2() {
                galleryImages2[currentImageIndex2].classList.remove('active');
                currentImageIndex2 = (currentImageIndex2 + 1) % galleryImages2.length;
                galleryImages2[currentImageIndex2].classList.add('active');
            }

            setInterval(changeImage2, 3000); // Cambia de imagen cada 3 segundos

            // Galería de imágenes para la sección 3
            const galleryImages3 = document.querySelectorAll('#unique-section3 .gallery img');
            let currentImageIndex3 = 0;

            function changeImage3() {

                galleryImages3[currentImageIndex3].classList.add('active');
            }

            setInterval(changeImage3, 3000); // Cambia de imagen cada 3 segundos

            // Fondo de galería de la sección 1
            const backgroundGalleryImages = document.querySelectorAll('.background-gallery img');
            let currentBackgroundImageIndex = 0;

            function changeBackgroundImage() {

                backgroundGalleryImages[currentBackgroundImageIndex].classList.add('active');
            }

            setInterval(changeBackgroundImage, 5000); // Cambia de imagen de fondo cada 5 segundos
        });
